cmake_minimum_required(VERSION 3.20)
project(kvsnap_dataplane LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(KVSNAP_CUDA "Build the real pinned-memory CUDA data plane" ON)
find_package(Threads REQUIRED)
add_library(kvsnap_io SHARED io.cpp)
target_link_libraries(kvsnap_io PRIVATE Threads::Threads)
target_compile_options(kvsnap_io PRIVATE -Wall -Wextra -Werror)
add_executable(kvsnap_io_test io_test.cpp)
target_link_libraries(kvsnap_io_test PRIVATE kvsnap_io)
enable_testing()
add_test(NAME io COMMAND kvsnap_io_test)
if(KVSNAP_CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(kvsnap_cuda SHARED cuda_pool.cpp)
  target_link_libraries(kvsnap_cuda PRIVATE CUDA::cudart)
  target_compile_options(kvsnap_cuda PRIVATE -Wall -Wextra -Werror)
endif()
